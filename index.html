export default { async fetch(request) { const url = new URL(request.url); const response = await fetch(request); // استخراج النطاق من الرابط const parts = url.pathname.split('/').filter(Boolean); let domain = null; for (let part of parts) { if (part.includes('.') && !part.endsWith('.html')) { domain = part.toLowerCase(); break; } } if (!domain) return response; const dynamicTitle = `${domain} | WHOIS, DNS Records & Valuation Report`; const dynamicDesc = `Detailed WHOIS lookup, DNS records, server location, and valuation report for ${domain}.`; // استبدال الـ Meta Tags في الـ HTML مباشرة قبل إرساله للفاحص return new HTMLRewriter() .on('title', { element(e) { e.setInnerContent(dynamicTitle); } }) .on('meta[name="description"]', { element(e) { e.setAttribute('content', dynamicDesc); } }) .on('meta[property="og:title"]', { element(e) { e.setAttribute('content', dynamicTitle); } }) .on('meta[property="og:description"]', { element(e) { e.setAttribute('content', dynamicDesc); } }) .on('meta[name="twitter:title"]', { element(e) { e.setAttribute('content', dynamicTitle); } }) .on('meta[name="twitter:description"]', { element(e) { e.setAttribute('content', dynamicDesc); } }) .transform(response); } };
